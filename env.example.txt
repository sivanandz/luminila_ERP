# ===========================================
# SINGLE ACCOUNT (default deployment model)
# ===========================================
# This installation serves one account, Niththigal: one PocketBase database and
# one account identity for every record. Full guide:
# docs/SINGLE_ACCOUNT_DEPLOYMENT_2026-10-29.md
#
# TENANT_MODE — "single" (default) or "multi". Leave unset for the shipped
# single-account deployment; set "multi" only to opt back into one database per
# login email.
# TENANT_MODE=single
#
# Re-enabling multi-account needs BOTH mode variables set to "multi" AND a
# client rebuild: NEXT_PUBLIC_* values are baked into the browser bundle at
# build time, so setting NEXT_PUBLIC_TENANT_MODE at runtime leaves the client
# in single mode.
# NEXT_PUBLIC_TENANT_MODE=multi
#
# The one account identity every row belongs to. ACCOUNT_EMAIL is server-only
# and NEXT_PUBLIC_ACCOUNT_EMAIL is the copy that reaches the browser (login
# prefill, client-side stamps). Set both to the SAME address whenever you set
# either, otherwise the server and the browser disagree about the owner.
# PocketBase compares a row's account_email with the signed-in user, so the
# operator must log in as this address; any other login is rejected on create.
# Default when both are unset: niththigal@gmail.com
# ACCOUNT_EMAIL=niththigal@gmail.com
# NEXT_PUBLIC_ACCOUNT_EMAIL=niththigal@gmail.com
#
# First-run admin, created by the boot hook only when the users collection is
# empty (single mode has no setup wizard). INITIAL_ADMIN_EMAIL is honoured only
# when it equals the single account identity above; otherwise boot logs a
# warning and keeps SINGLE_ACCOUNT_EMAIL. Use a strong password.
# INITIAL_ADMIN_EMAIL=niththigal@gmail.com
# INITIAL_ADMIN_PASSWORD=change-me-before-first-boot
